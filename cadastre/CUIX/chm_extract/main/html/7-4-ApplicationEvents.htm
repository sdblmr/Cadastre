Message Events

Message events refer to events triggered at the software level, such as keyboard/mouse input, software exit, or system variable changes.

Common Message Events

Event Function Overview
PreTranslateMessage Intercepts keyboard and mouse input events
SystemVariableChanged System variable modification event
BeginQuit Software exit event; can prevent closing via IsVetoed property
BeginDoubleClick Double-click event

Adding and Removing Message Events

Adding Events

    static public void AddAppEvents()
    {
        Database db = HostApplicationServices.WorkingDatabase;
        Document doc = Application.DocumentManager.GetDocument(db);
        Editor ed = doc.Editor;

        try
        {
            Application.PreTranslateMessage += Application_PreTranslateMessage;
            Application.SystemVariableChanged += Application_SystemVariableChanged;
            Application.BeginQuit += Application_BeginQuit;
            Application.BeginDoubleClick += Application_BeginDoubleClick;
        }
        catch (System.Exception ex)
        {
            Application.ShowAlertDialog(ex.ToString());
        }
        ed.WriteMessage("\nApplication Events Added!");
    }

Removing Events

    static public void RemoveAppEvents()
    {
        Database db = HostApplicationServices.WorkingDatabase;
        Document doc = Application.DocumentManager.GetDocument(db);
        Editor ed = doc.Editor;

        try
        {
            Application.PreTranslateMessage -= Application_PreTranslateMessage;
            Application.SystemVariableChanged -= Application_SystemVariableChanged;
            Application.BeginQuit -= Application_BeginQuit;
            Application.BeginDoubleClick -= Application_BeginDoubleClick;
        }
        catch (System.Exception ex)
        {
            Application.ShowAlertDialog(ex.ToString());
        }
        ed.WriteMessage("\nApplication Events Removed!");
    }

Defining Event Content

Keyboard and Mouse Input Events

In this event, PreTranslateMessageEventArgs contains the current mouse/keyboard input state. Use "message" to get the current operation and "wParam" to retrieve the input value. 

    const int WM_CHAR = 258;
    private static void Application_PreTranslateMessage(object sender, PreTranslateMessageEventArgs e)
    {
        // Check if it's a lowercase letter input
        if (e.Message.message == WM_CHAR &&
            (e.Message.wParam.ToInt32() >= 97 && e.Message.wParam.ToInt32() <= 122))
        {
            MSG msg = e.Message;
            // Convert lowercase to uppercase
            msg.wParam = (IntPtr)(e.Message.wParam.ToInt32() - 32);
            e.Message = msg;
        }
    }

Application Events

    private static void Application_BeginQuit(object sender, BeginQuitEventArgs e)
    {
        var res = System.Windows.Forms.MessageBox.Show("Are your sure to quit?", "Alert", System.Windows.Forms.MessageBoxButtons.YesNo, System.Windows.Forms.MessageBoxIcon.Warning);
        if (res == System.Windows.Forms.DialogResult.No)
        {
            // Set to true to prevent application exit
            e.IsVetoed = true;
        }
    }

    private static void Application_SystemVariableChanged(object sender, ZwSoft.ZwCAD.ApplicationServices.SystemVariableChangedEventArgs e)
    {
        if (e.Changed)
        {
            WriteMsg($"\nSystem variable {e.Name} was changed.");
        }
    }
    private static void Application_BeginDoubleClick(object sender, BeginDoubleClickEventArgs e)
    {
        WriteMsg($"\nDouble click at {e.Location}");
    }
