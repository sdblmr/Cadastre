CUIX Interface

The CUIX file is a menu file used to store customized interface layouts. Starting from 2025, ZWCAD supports a series of APIs to manipulate CUIX files, allowing rapid customization of CUIX files.

Preparation

First, add a reference to ZcCui.dll in your project and include the following in your code:

    using ZwSoft.ZwCAD.Customization;

Loading and Unloading CUIX Files

A prepared CUIX file can be loaded as a partial customization file under the current main CUIX.

Load a CUIX File

    string dllPath = System.IO.Path.GetDirectoryName(Assembly.GetExecutingAssembly().Location);
    string cuixFilePath = dllPath + "\\..\\..\\Sample25.cuix";
    if (System.IO.File.Exists(cuixFilePath))
    {
        Application.LoadPartialMenu(cuixFilePath);
    }

Unload a CUIX File

    // Get the main CUIX object
    string sMainCuiFile = (string)Application.GetSystemVariable("MENUNAME");
    sMainCuiFile += ".cuix";
    CustomizationSection mainMenu = new CustomizationSection(sMainCuiFile);

    string unloadCuiName = "Sample25.cuix";
    if (mainMenu.PartialCuiFiles.Contains(unloadCuiName))
    {
        Application.UnloadPartialMenu(unloadCuiName);
    }

Create a CUIX File

In addition to loading existing CUIX files, you can dynamically generate new CUIX files through code. This reduces the number of files required in the plugin package and allows pre-checking for existing menu groups in CAD to avoid conflicts caused by loading duplicate files. 

    static public void CreateCuixFile()
    {
        string menuGroupName = "MyCui";
        string cuiFileName = Environment.GetFolderPath(Environment.SpecialFolder.Desktop) + $"\\{menuGroupName}.cuix";
        CustomizationSection cs = AddCui(cuiFileName, menuGroupName);

        EditCuixFile(cs);

        string sMainCuiFile = (string)Application.GetSystemVariable("MENUNAME");
        sMainCuiFile += ".cuix";
        CustomizationSection mainMenu = new CustomizationSection(sMainCuiFile);

        // Unload existing CUIX with the same name before reloading to ensure updates are displayed
        if (mainMenu.PartialCuiFiles.Contains(cuiFileName))
        {
            Application.UnloadPartialMenu(cuiFileName);
        }
        Application.LoadPartialMenu(cuiFileName);
    }
        
    // Create CUIX
    static public CustomizationSection AddCui(string cuiFileName, string menuGroupName)
    {
        CustomizationSection cs;
        if (!System.IO.File.Exists(cuiFileName))
        {
            cs = new CustomizationSection
            {
                MenuGroupName = menuGroupName,
            };
            cs.SaveAs(cuiFileName);
        }
        else
        {
            cs = new CustomizationSection(cuiFileName);
        }
        return cs;
    }

Edit CUIX File

After creating a CUIX file, you can add custom macros and link them to menu items like dropdown menus or toolbars.

    static public void EditCuixFile(CustomizationSection cs)
    {
        string dllPath = System.IO.Path.GetDirectoryName(Assembly.GetExecutingAssembly().Location);
        // Add command macro
        cs.AddMacro("MyLine", "^C^C_Line", "ID_MyLine", "Create new Line: Line", dllPath + "\\..\\..\\Resources\\a_large.png");
        StringCollection sc = new StringCollection() { "MyPopMenu" };
        // Add dropdown menu
        PopMenu popMenu = cs.MenuGroup.AddPopMenu("My Menu", sc, "ID_MyMenu");
        popMenu.AddMenuItem(-1, "MyLine", "ID_MyLine");
        // Save modified CUIX file
        cs.Save();
    }

Add Macro Command

Encapsulate an AddMacro method to create a new macro in the specified macro group using CreateMenuMacro API:

    // name: Command name, command: Macro, tag: Element ID, helpString: Description, imagePath: Image
    static public MenuMacro AddMacro(this CustomizationSection cs, string name, string command, string tag, string helpString, string imagePath)
    {
        MenuGroup menuGroup = cs.MenuGroup;
        MacroGroup macroGroup = menuGroup.FindMacroGroup(menuGroup.Name);
        if (macroGroup == null)
        {
            macroGroup = new MacroGroup(menuGroup.Name, menuGroup);
        }

        foreach (MenuMacro macro in macroGroup.MenuMacros)
        {
            if (macro.ElementID == tag)
            {
                return null;
            }
        }

        MenuMacro menuMacro = macroGroup.CreateMenuMacro(name, command, tag);
        menuMacro.macro.HelpString = helpString;
        menuMacro.macro.LargeImage = menuMacro.macro.SmallImage = imagePath;
        return menuMacro;
    }

Add Dropdown Menu and Link to Macro

Create a dropdown menu and bind it to the macro using the element ID (tag) created earlier.

    static public PopMenu AddPopMenu(this MenuGroup menuGroup, string name, StringCollection aliasList, string tag)
    {
        PopMenu pm = null;
        if (menuGroup.PopMenus.IsNameFree(name))
        {
            pm = new PopMenu(name, aliasList, tag, menuGroup);
        }
        else
        {
            pm = menuGroup.PopMenus.FindPopWithName(name);
        }
        return pm;
    }
    static public PopMenuItem AddMenuItem(this PopMenu parentMenu, int index, string name, string macroId)
    {
        PopMenuItem menuItem = null;
        foreach (PopMenuItem pmi in parentMenu.PopMenuItems)
        {
            if (pmi.Name == name) return menuItem;
        }
        menuItem = new PopMenuItem(parentMenu, index);
        if (name != null)
        {
            menuItem.Name = name;
        }
        menuItem.MacroID = macroId;
        return menuItem;
    }
