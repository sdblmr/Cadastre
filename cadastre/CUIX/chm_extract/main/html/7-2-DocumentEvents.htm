Document Events

Document events are used to receive special events at the CAD software level, such as opening or closing drawings, executing command processes, etc.

Common Document Events

Event Function Overview
BeginDwgOpen Document about to be opened
BeginDocumentClose Document about to be closed
CommandWillStart Command about to be executed
CommandEnded Command execution completed
CommandCancelled Command cancelled

Adding and Removing Document Events

Adding Events

    static public void AddDocEvents()
    {
        Database db = HostApplicationServices.WorkingDatabase;
        Document doc = Application.DocumentManager.GetDocument(db);
        Editor ed = doc.Editor;

        try
        {
            doc.BeginDwgOpen += Doc_BeginDwgOpen;
            doc.BeginDocumentClose += Doc_BeginDocumentClose;
            doc.CommandWillStart += Doc_CommandWillStart;
            doc.CommandEnded += Doc_CommandEnded;
            doc.CommandCancelled += Doc_CommandCancelled;
        }
        catch (Exception ex)
        {
            Application.ShowAlertDialog(ex.ToString());
            return;
        }

        ed.WriteMessage("\nDocument Events Added!");
    }

Removing Events

    static public void RemoveDocEvents()
    {
        Database db = HostApplicationServices.WorkingDatabase;
        Document doc = Application.DocumentManager.GetDocument(db);
        Editor ed = doc.Editor;

        try
        {
            doc.BeginDwgOpen -= Doc_BeginDwgOpen;
            doc.BeginDocumentClose -= Doc_BeginDocumentClose;
            doc.CommandWillStart -= Doc_CommandWillStart;
            doc.CommandEnded -= Doc_CommandEnded;
            doc.CommandCancelled -= Doc_CommandCancelled;
        }
        catch (Exception ex)
        {
            Application.ShowAlertDialog(ex.ToString());
            return;
        }

        ed.WriteMessage("\nDocument Events Removed!");
    }

Defining Event Content

In the event handler function, the second parameter generally contains information such as the document or command name that triggered the operation. 

    private static void Doc_BeginDocumentClose(object sender, DocumentBeginCloseEventArgs e)
    {
        // The vote() method in DocumentBeginCloseEventArgs can be used to prevent document closure
        WriteMsg($"\nBeginDocumentClose");
    }

    private static void Doc_BeginDwgOpen(object sender, DrawingOpenEventArgs e)
    {
        WriteMsg($"\nBeginDwgOpen, filename: {e.FileName}");
    }

    private static void Doc_CommandCancelled(object sender, CommandEventArgs e)
    {
        WriteMsg($"\nCommandCancelled, command: {e.GlobalCommandName}");           
    }

    private static void Doc_CommandEnded(object sender, CommandEventArgs e)
    {
        WriteMsg($"\nCommandEnded, command: {e.GlobalCommandName}");           
    }

    private static void Doc_CommandWillStart(object sender, CommandEventArgs e)
    {
        WriteMsg($"\nCommandWillStart, command: {e.GlobalCommandName}");           
    }
