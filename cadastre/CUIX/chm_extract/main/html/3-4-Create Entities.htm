Methods for Creating and Editing Common Entities

This chapter introduces the creation methods for common entities such as Line, Circle (Arc), Ellipse (Arc), Polyline, Text, Hatch, Table, Dimension, etc.

2. Edit a Line

C#

    [CommandMethod("ModLine", CommandFlags.Modal)]
    public void ModLine()
    {
        Document objDoc = Application.DocumentManager.MdiActiveDocument;
        Database objDB = objDoc.Database;
        using (Transaction trans = objDoc.TransactionManager.StartTransaction())
        {
            BlockTable zcBLT = (BlockTable)trans.GetObject(objDB.BlockTableId, OpenMode.ForRead);

            foreach (ObjectId entOid in zcBlkTblRec) // Traverse all entities in model space
            {
                Entity ent = trans.GetObject(entOid, OpenMode.ForWrite) as Entity;
                if (ent != null && ent.GetType() == typeof(Line)) // Filter line objects
                {
                    Line line = ent as Line;
                    if (line.ColorIndex == 1) // Check line color
                    {
                        Point3d startPoint = new Point3d(0, 100, 0);
                        Point3d endPoint = new Point3d(100, 0, 0);
                        line.StartPoint = startPoint; // Modify start point
                        line.EndPoint = endPoint;     // Modify end point
                    }
                    else
                    {
                        line.Erase(); // Delete the line
                    }
                }
            }
            trans.Commit();
        }
    }

1. Create Ellipse

    [CommandMethod("DrawEllipse", CommandFlags.Modal)]

        }
    }

2. Modify Ellipse

C#

 
    [CommandMethod("ModEllipse", CommandFlags.Modal)] 
    public void ModEllipse() 
    { 
        Document objDoc = Application.DocumentManager.MdiActiveDocument; Database objDB = objDoc.Database;
        using (Transaction trans = objDoc.TransactionManager.StartTransaction())
        {
            BlockTable blockTable = trans.GetObject(objDB.BlockTableId, OpenMode.ForRead) as BlockTable;
            BlockTableRecord modelSpace = trans.GetObject(
                blockTable[BlockTableRecord.ModelSpace], 
                OpenMode.ForWrite) as BlockTableRecord;
        
            foreach (ObjectId entId in modelSpace)
            {
                Entity ent = trans.GetObject(entId, OpenMode.ForWrite) as Entity;
                if (ent is Ellipse ellipse) // Filter ellipse objects
                {
                    // Modification condition: Ellipse with angle range less than 360 degrees
                    if (ellipse.EndAngle - ellipse.StartAngle < 2 * Math.PI) 
                    {
                        // Modify to a full ellipse
                        ellipse.EndAngle = ellipse.StartAngle + 2 * Math.PI; 
                        ellipse.ColorIndex = 3; // Change to green
                    }
                    else 
                    {
                        // Modify shape parameters
                        ellipse.RadiusRatio = 0.3;               // Adjust the ratio of major to minor axes
                    }
                }
            }
            trans.Commit();
        }
    }

1. Create Polyline

C#

 
        [CommandMethod("DrawPolyline", CommandFlags.Modal)]

        {
            // Get block table (opened in read mode)
            BlockTable blkTbl = trans.GetObject(db.BlockTableId, OpenMode.ForRead) as BlockTable;
            // Get model space block table record (opened in write mode)
            BlockTableRecord blkTblRec = trans.GetObject(blkTbl[BlockTableRecord.ModelSpace], OpenMode.ForWrite) as BlockTableRecord;
    
            // Prompt user to select objects
            PromptSelectionResult selRes = ed.GetSelection();
            if (selRes.Status != PromptStatus.OK)

            // Get selection set
            SelectionSet sSet = selRes.Value;
            // Get object IDs from the selection set
            ObjectId[] objIds = sSet.GetObjectIds();
    
            // Create a dictionary to store entities and their areas
            Dictionary<ObjectId, double> dic1 = new Dictionary<ObjectId, double>();
            foreach (ObjectId id in objIds)
            {
                // Get entity object (opened in write mode)
                Entity entity = (Entity)id.GetObject(OpenMode.ForWrite);
                // Calculate entity area and store in dictionary
                dic1.Add(id, ((Curve)entity).Area);
            }
    
            // Sort the dictionary by area in ascending order
            Dictionary<ObjectId, double> dic2 = dic1.OrderBy(o => o.Value).ToDictionary(k => k.Key, v => v.Value);
    
            // Create a new Hatch object
            using (Hatch hatch = new Hatch())
            {
                // Add the hatch object to the model space block table record
                blkTblRec.AppendEntity(hatch);
                // Add the newly created DBObject to the transaction
                trans.AddNewlyCreatedDBObject(hatch, true);
    
                // Set hatch properties
                hatch.Normal = Vector3d.ZAxis; // Set hatch normal direction
                hatch.Elevation = 0.0; // Set hatch elevation
                    hatch.Associative = true; // Set hatch as associative
                    hatch.SetHatchPattern(HatchPatternType.PreDefined, "SOLID"); // Set hatch pattern to SOLID
        
                    // Prompt the user to select the hatch style
                    PromptKeywordOptions keyOpts = new PromptKeywordOptions("");
                    keyOpts.Message = "Style type [Normal(N)/Outer(O)/Ignore(I)]:";
                    keyOpts.Keywords.Add("N");
                    keyOpts.Keywords.Add("O");
                    keyOpts.Keywords.Add("I");
                    keyOpts.AllowNone = false;
                    PromptResult keyRes = doc.Editor.GetKeywords(keyOpts);
        
                    // Set the hatch style based on user input
                    if (keyRes.StringResult == "N")
                        hatch.HatchStyle = HatchStyle.Normal;
                    else if (keyRes.StringResult == "O")
                        hatch.HatchStyle = HatchStyle.Outer;
                    else if (keyRes.StringResult == "I")
                        hatch.HatchStyle = HatchStyle.Ignore;
        
                    // Add hatch boundaries (loops) based on the order in the dictionary
                    int nDep = 0;
                    foreach (KeyValuePair<ObjectId, double> item in dic2)
                    {
                        if (nDep == dic2.Count - 1)
                        {
                            // The last entity, added as an external loop
                            ObjectIdCollection externalIdColl = new ObjectIdCollection();
                            externalIdColl.Add(item.Key);
                            hatch.AppendLoop(HatchLoopTypes.External, externalIdColl);
                        }
                        else if (nDep == dic2.Count - 2)
                        {
                            // The second-to-last entity, added as the outermost loop
                            ObjectIdCollection outerMostIdColl = new ObjectIdCollection();
                            outerMostIdColl.Add(item.Key);
                            hatch.AppendLoop(HatchLoopTypes.Outermost, outerMostIdColl);
                        }
                        else

                            // Other entities, added as default loops
                            ObjectIdCollection defaultIdColl = new ObjectIdCollection();
                            defaultIdColl.Add(item.Key);
                            hatch.AppendLoop(HatchLoopTypes.Default, defaultIdColl);
                        }
                        nDep++;
                    }
        
                    // Generate the hatch graphics
                    hatch.EvaluateHatch(true);
                }
                trans.Commit();
            }

    {

        foreach(ObjectId id in modelSpace)
        {
            Entity ent = trans.GetObject(id, OpenMode.ForWrite) as Entity;
            if(ent is Dimension dim)
            {
                // Uniformly modify dimension colors
                dim.ColorIndex = 2; // Change to yellow
                
                // Modify text prefix
                if(dim is RadialDimension radial)
                    radial.DimensionText = "Radius " + radial.Measurement;
            }
            else if(ent is MLeader leader)
            {
                leader.TextHeight *= 1.5; // Enlarge text by 150%
            }
        }
        trans.Commit();
    }
