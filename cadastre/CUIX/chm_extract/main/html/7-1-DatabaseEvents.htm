Database Events

Database events are used to receive notifications related to database state changes, such as objects being added to the database, or objects being deleted, modified, etc. The sender of these events is the database itself, so all such events we define will be added to the database event list.

Common Database Events

Event Function Overview
ObjectAppended Entity added
ObjectErased Entity deleted
ObjectModified Entity modified
ObjectOpenedForModify Entity opened for modification

Adding and Removing Database Events

Adding Events

    static public void AddDBEvents()
    {
        Database db = HostApplicationServices.WorkingDatabase;
        Document doc = Application.DocumentManager.GetDocument(db);
        Editor ed = doc.Editor;

        try
        {
            db.ObjectAppended += Db_ObjectAppended;
            db.ObjectErased += Db_ObjectErased;
            db.ObjectModified += Db_ObjectModified;
            db.ObjectOpenedForModify += Db_ObjectOpenedForModify;
        }
        catch (Exception ex)
        {
            Application.ShowAlertDialog(ex.ToString());
            return;
        }

        ed.WriteMessage("\nDatabase Events Added!");
    }

Removing Events

    static public void RemoveDBEvents()
    {
        Database db = HostApplicationServices.WorkingDatabase;
        Document doc = Application.DocumentManager.GetDocument(db);
        Editor ed = doc.Editor;

        try
        {
            db.ObjectAppended -= Db_ObjectAppended;
            db.ObjectErased -= Db_ObjectErased;
            db.ObjectModified -= Db_ObjectModified;
            db.ObjectOpenedForModify -= Db_ObjectOpenedForModify;
        }
        catch (Exception ex)
        {
            Application.ShowAlertDialog(ex.ToString());
            return;
        }

        ed.WriteMessage("\nDatabase Events Removed!");
    }

Defining Event Handlers

In the event handler function, the first parameter "sender" is the database object that triggered the event. The second parameter "e" contains event arguments, which provide specific information such as the added entity object, deleted entity object, etc. 

    private static void Db_ObjectOpenedForModify(object sender, ObjectEventArgs e)
    {
        WriteMsg($"\nObjectOpenedForModify, Handle: {e.DBObject.Handle.ToString()}");
    }

    private static void Db_ObjectModified(object sender, ObjectEventArgs e)
    {
        WriteMsg($"\nObjectModified, Handle: {e.DBObject.Handle.ToString()}");
    }

    private static void Db_ObjectErased(object sender, ObjectErasedEventArgs e)
    {
        WriteMsg($"\nObjectErased, Handle: {e.DBObject.Handle.ToString()}");
    }

    private static void Db_ObjectAppended(object sender, ObjectEventArgs e)
    {
        WriteMsg($"\nObjectAppended, Handle: {e.DBObject.Handle.ToString()}");
    }
