Display Redefinition

The DrawableOverrule class allows overriding the display of existing base entity objects to render them in different shapes.

Creating a Custom DrawableOverrule

Create a new class file named "MyEntOverrule" in the project, modify the code to inherit from the DrawableOverrule class, and override the WorldDraw virtual method.

    class MyEntOverrule : DrawableOverrule
    {
        public override bool WorldDraw(Drawable drawable, WorldDraw wd) 
        {
            ...
        }
    }
Interface Parameters Overview
drawable The object to be overridden, i.e., the original entity to which the overrule is applied.
wd Used to render the final display graphics, which will not be added to the model space.

Rendering Implementation

After obtaining the original object to be overridden, create custom graphics based on the object's stored data and render them via the WorldDraw object.

In this example, we use data stored in the entity's XData along with the circle's center and radius to draw inner and outer circles.

    public override bool WorldDraw(Drawable drawable, WorldDraw wd)
    {
        // Check if it's a Circle
        if (drawable is Circle myCircle)
        {
            // Retrieve XData from the circle. Commands.m_xdata is a constant
            ResultBuffer rb = myCircle.GetXDataForApplication(Commands.m_xdata);
            TypedValue[] values = rb.AsArray();
            int spacing = (int)values[1].Value;

            double radius = myCircle.Radius;
            Point3d center = myCircle.Center;

            // Set the color of subsequent graphics to red. To set different colors, call this interface multiple times before rendering.
            wd.SubEntityTraits.Color = 1;
            wd.Geometry.Circle(center, radius - spacing / 2, Vector3d.ZAxis);

            wd.SubEntityTraits.Color = 2;
            // For shapes not supported by Geometry, use the entity's WorldDraw method
            Circle cir = new Circle(center, Vector3d.ZAxis, radius + spacing / 2);
            cir.WorldDraw(wd);

            // Return true to render only the custom graphics. To render the original entity, use base.WorldDraw
            return true;
        }
        return base.WorldDraw(drawable, wd);
    }

Adding and Removing Overrules

Use Overrule.AddOverrule to apply the overrule to a specific entity type. Use SetXDataFilter to ensure the overrule applies only to entities with matching XData.

Adding the Overrule

    public const string m_xdata = "MY_XDATA";
    static MyEntOverrule drawableOverride;

    [CommandMethod("AddOverrule")]
    static public void AddDrawableOverrule()
    {
        Editor ed = Application.DocumentManager.MdiActiveDocument.Editor;

        if (drawableOverride == null)
        {
            drawableOverride = new MyEntOverrule();
            // Add overrule to Circle type
            Overrule.AddOverrule(RXObject.GetClass(typeof(Circle)), drawableOverride, false);
            // Add XData filter to apply only to entities with this XData
            drawableOverride.SetXDataFilter(m_xdata);
        }

        Overrule.Overruling = true;
        ed.Regen();

        ed.WriteMessage("\nOverrule Added!");
    }

Removing the Overrule

    [CommandMethod("RemoveOverrule")]
    static public void RemoveDrawableOverrule()
    {
        Editor ed = Application.DocumentManager.MdiActiveDocument.Editor;

        if (drawableOverride != null)
        {
            Overrule.RemoveOverrule(RXObject.GetClass(typeof(Circle)), drawableOverride);
            drawableOverride = null;
        }

        Overrule.Overruling = false;
        ed.Regen();

        ed.WriteMessage("\nOverrule Removed!");
    }

Adding XData to Entities

Since the overrule uses an XData filter, you must first add XData with the specified name to target entities and store user input values in the XData.

    [CommandMethod("AddData")]
    static public void AddData()
    {
        Document doc = Application.DocumentManager.MdiActiveDocument;
        Database db = doc.Database;
        Editor ed = doc.Editor;

        // Select a Circle
        PromptEntityOptions opt = new PromptEntityOptions("\nSelect a Circle:");
        opt.SetRejectMessage("\nPlease select a Circle!");
        opt.AddAllowedClass(typeof(Circle), true);

        PromptEntityResult optRes = ed.GetEntity(opt);
        if (optRes.Status != PromptStatus.OK)
            return;

        // Get user input
        PromptIntegerOptions intOpt = new PromptIntegerOptions("\nEnter the spacing:");
        // Minimum input value is 0
        intOpt.LowerLimit = 0;

        using (var tr = doc.TransactionManager.StartTransaction())
        {
            Circle ent = tr.GetObject(optRes.ObjectId, OpenMode.ForWrite) as Circle;
            // Ensure spacing does not exceed twice the radius to avoid negative inner radius
            intOpt.UpperLimit = (int)ent.Radius * 2;

            RegAppTable regTable = tr.GetObject(db.RegAppTableId, OpenMode.ForWrite) as RegAppTable;
            // Check if the XData name is already registered
            if (!regTable.Has(m_xdata))
            {
                RegAppTableRecord regRec = new RegAppTableRecord();
                regRec.Name = m_xdata;
                regTable.Add(regRec);
                tr.AddNewlyCreatedDBObject(regRec, true);
                intOpt.DefaultValue = 10;
            }
            else
            {
                ResultBuffer rb = ent.GetXDataForApplication(m_xdata);
                intOpt.DefaultValue = (int)rb.AsArray()[1].Value;
            }

            PromptIntegerResult intRes = ed.GetInteger(intOpt);
            int spacing = intOpt.DefaultValue;
            if (intRes.Status == PromptStatus.OK)
            {
                spacing = intRes.Value;
            }
            // Write XData
            ent.XData = new ResultBuffer(
                new TypedValue((int)DxfCode.ExtendedDataRegAppName, m_xdata),
                new TypedValue((int)DxfCode.ExtendedDataInteger32, spacing));

            tr.Commit();
        }
        ed.Regen();
    }

Implementation Effect

Note: The execution of AddOverrule and AddData commands does not need to distinguish the order.
