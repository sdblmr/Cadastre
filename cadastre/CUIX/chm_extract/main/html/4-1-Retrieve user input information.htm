Retrieve User Input Information

Description

This section explains how to interact with users via the command line to retrieve input information, such as real numbers, strings, keywords, etc.

User Input Functions

The table below lists functions for retrieving user input. Subsequent articles will demonstrate their usage.

Function Name Purpose Prompt Options Class Prompt Result Class
GetDouble() Prompts the user to input a real number PromptDoubleOptions PromptDoubleResult
GetInteger() Prompts the user to input an integer PromptIntegerOptions PromptIntegerResult
GetKeywords() Prompts the user to input a keyword PromptKeywordOptions PromptResult 
GetString() Prompts the user to input a string PromptStringOptions PromptResult

GetDouble() - Retrieve a Real Number

public void TestGetDouble()
{
    Editor ed = Application.DocumentManager.MdiActiveDocument.Editor;
    PromptDoubleOptions dblOpts = new PromptDoubleOptions("Input a real number:");

    dblOpts.AllowNone = false;
    dblOpts.AllowZero = false;
    dblOpts.DefaultValue = 0.0;

    PromptDoubleResult dblRes = ed.GetDouble(dblOpts);
    if (dblRes.Status == PromptStatus.OK)
    {
        ed.WriteMessage("\nThe real number is {0}", dblRes.Value);
    }
    else if (dblRes.Status == PromptStatus.Cancel)
    {
        ed.WriteMessage("\nYou have cancelled!");
    }
    else
    {
        ed.WriteMessage("\nOther status: " + dblRes.Status.ToString());
    }
}

Execution Result:

When executing the TestGetDouble command, the user is prompted with "Input a real number:". If the user enters 10.2, the command line outputs "The real number is 10.2".

GetInteger() - Retrieve an Integer

[CommandMethod("TestGetInteger")]
public void TestGetInteger()
{
    Editor ed = Application.DocumentManager.MdiActiveDocument.Editor;
    PromptIntegerOptions intOpts = new PromptIntegerOptions("Input an Integer:");
    intOpts.AllowNegative = false;
    intOpts.AllowNone = false;
    intOpts.AllowZero = false;
    intOpts.DefaultValue = 100;
    PromptIntegerResult IntRes = ed.GetInteger(intOpts);
    if (IntRes.Status == PromptStatus.OK)
    {
        ed.WriteMessage("\nThe Integer is {0}", IntRes.Value);
    }
    else if (IntRes.Status == PromptStatus.Cancel)
    {
        ed.WriteMessage("\nYou have cancelled!");
    }
    else
    {
        ed.WriteMessage("\nOther status: " + IntRes.Status.ToString());
    }
}

Execution Result:

When executing the TestGetKeyWord command, the user is prompted with "Input an Integer:". If the user enters 10, the command line outputs "The Integer is 10".

GetKeywords() - Retrieve a Keyword

[CommandMethod("TestGetKeyWord")]
public void TestGetKeyWord()
{
    Editor ed = Application.DocumentManager.MdiActiveDocument.Editor;

    PromptKeywordOptions kwOpts = new PromptKeywordOptions("Input KeyWord ");

    kwOpts.Keywords.Add("A", "A", "Branch A(A)", true, true);
    kwOpts.Keywords.Add("B", "B", "Branch B(B)", true, true);
    kwOpts.Keywords.Add("C", "C", "Branch C(C)", true, true);
    kwOpts.Keywords.Add("D", "D", "Branch D(D)", true, true);
    kwOpts.Keywords.Default = "D";

    PromptResult kwordRes = ed.GetKeywords(kwOpts);

    if (kwordRes.Status == PromptStatus.OK)
    {
        switch (kwordRes.StringResult)
        {
            case "A":
                ed.WriteMessage("This is Branch A");
                break;
            case "B":
                ed.WriteMessage("This is Branch B");
                break;
            case "C":
                ed.WriteMessage("This is Branch C");
                break;
            case "D":
                ed.WriteMessage("This is Branch D");
                break;
        }
    }
    else
    {
        ed.WriteMessage("Other Status: " + kwordRes.Status.ToString());
    }
}

Execution Result:

When executing the TestGetKeyWord command, the user is prompted with "Input KeyWord [Branch A(A)/Branch B(B)/Branch C(C)/Branch D(D)]:". If the user enters B, the command line outputs "This is Branch B".

GetString() - Retrieve a String

[CommandMethod("TestGetString")]
public void TestGetString()
{
    Editor ed = Application.DocumentManager.MdiActiveDocument.Editor;
    PromptStringOptions strOpts = new PromptStringOptions("Input a string: ");
    strOpts.AllowSpaces = false;     //Spaces are not allowed
    PromptResult strRes = ed.GetString(strOpts);
    if (strRes.Status == PromptStatus.OK)
    {
        ed.WriteMessage("The string is " + strRes.StringResult);
    }
    else if (strRes.Status == PromptStatus.Cancel)
    {
        ed.WriteMessage("\nYou have cancelled!");
    }
    else
    {
        ed.WriteMessage("\nOther status: " + strRes.Status.ToString());
    }
}

Execution Result:

When executing the TestGetString command, the user is prompted to input a string "Input a string:". If the user enters "hello", the command line will output "The string is hello".
